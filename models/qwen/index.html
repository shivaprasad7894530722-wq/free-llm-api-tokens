Example request

This example uses the published base URL for Groq. Replace MODEL_ID with the model id from that provider’s official docs. We do not guess the id.

import os
from openai import OpenAI

client = OpenAI(
    api_key=os.environ["API_KEY"],
    base_url="https://api.groq.com/openai/v1",
)

resp = client.chat.completions.create(
    model="MODEL_ID",  # copy the model id from the provider's docs
    messages=[{"role": "user", "content": "Hello"}],
)
print(resp.choices[0].message.content)
